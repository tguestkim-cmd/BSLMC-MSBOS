function rbcDisplay(rbcs) {

  if (rbcs === "TNS") return { val: "TNS", cls: "blood-value tns-val" };

  return { val: rbcs + " u", cls: "blood-value" };

function highlight(text, q) {

  if (!q) return text;

  const re = new RegExp(`(${q.replace(/[.*+?^${}()|[\]\\]/g,'\\$&')})`, "gi");

  return text.replace(re, '<mark style="background:rgba(59,130,246,0.35);color:#fff;border-radius:2px;padding:0 2px">$1</mark>');

}


function matches(item, q) {

  if (!q) return true;

  const s = q.toLowerCase();

  return (

    item.proc.toLowerCase().includes(s) ||

    item.epic.some(e => e.toLowerCase().includes(s)) ||

    item.dept.toLowerCase().includes(s) ||

    item.catLabel.toLowerCase().includes(s) ||

    item.conditions.toLowerCase().includes(s)

  );

}


// ═══════════════════════════════════════════════════════

// RENDER

// ═══════════════════════════════════════════════════════

function render() {

  const grid = document.getElementById("results-grid");

  const countEl = document.getElementById("results-count");

  grid.innerHTML = "";


  const filtered = DATA.filter(item => {

    if (activeDept !== "all" && item.dept !== activeDept) return false;

    if (activeCat  !== "all" && item.cat  !== activeCat)  return false;

    if (!matches(item, searchQ)) return false;

    return true;

  });


  countEl.innerHTML = `<span>${filtered.length}</span> procedure${filtered.length !== 1 ? "s" : ""} found`;


  if (filtered.length === 0) {

    grid.innerHTML = `<div class="empty-state">

      <div class="big-icon">🩸</div>

      <p>No procedures match your search.<br>Try different keywords or clear your filters.</p>

    </div>`;

    return;

  }


  filtered.forEach((item, idx) => {

    const d = rbcDisplay(item.rbcs);

    const card = document.createElement("div");

    card.className = "card";

    card.style.animationDelay = `${Math.min(idx, 20) * 20}ms`;

    card.dataset.idx = DATA.indexOf(item);


    const epicPreview = item.epic.slice(0, 2).map(e =>

      `<span class="epic-pill">${highlight(e, searchQ)}</span>`

    ).join("") + (item.epic.length > 2 ? `<span class="epic-pill">+${item.epic.length-2} more</span>` : "");


    card.innerHTML = `

      <div class="${barClass(item.cat)}"></div>

      <div class="card-body">

        <div class="card-top">

          <span class="dept-tag">${item.dept}</span>

          <span class="${catBadge(item.cat)}">${item.catLabel}</span>

        </div>

        <h3>${highlight(item.proc, searchQ)}</h3>

        <div class="blood-box">

          <div class="blood-item">

            <div class="blood-label">RBC Units</div>

            <div class="${d.cls}">${d.val}</div>

          </div>

          ${item.other ? `<div class="blood-item">

            <div class="blood-label">Additional</div>

            <div class="blood-value big">${item.other}</div>

          </div>` : ""}

        </div>

        ${epicPreview ? `<div class="epic-section">

          <div class="section-label">EPIC Code(s)</div>

          <div class="epic-codes">${epicPreview}</div>

        </div>` : ""}

        ${item.conditions ? `<div class="conditions">${highlight(item.conditions, searchQ)}</div>` : ""}

      </div>`;


    card.addEventListener("click", () => openModal(DATA.indexOf(item)));

    grid.appendChild(card);

  });

}


// ═══════════════════════════════════════════════════════

// MODAL

// ═══════════════════════════════════════════════════════

function openModal(idx) {

  const item = DATA[idx];

  const d = rbcDisplay(item.rbcs);

  document.getElementById("modal-title").textContent = item.proc;


  // Parse other products for display

  let extraCards = "";

  if (item.other) {

    const parts = item.other.split("+").map(s => s.trim()).filter(Boolean);

    parts.forEach(part => {

      const m = part.match(/^(\d+)\s+(.+)$/);

      if (m) {

        extraCards += `<div class="modal-blood-card">

          <div class="big-val" style="color:var(--muted)">${m[1]}</div>

          <div class="val-label">${m[2]}</div>

        </div>`;

      } else {

        extraCards += `<div class="modal-blood-card">

          <div class="big-val" style="font-size:1rem;color:var(--muted)">${part}</div>

          <div class="val-label">Additional</div>

        </div>`;

      }

    });

  }


  const rbcColor = { tns:"var(--tns)", c2:"var(--c2)", c4:"var(--c4)", c6:"var(--c6)", c8:"var(--c8)", tx:"var(--tx)" }[item.cat];


  document.getElementById("modal-body").innerHTML = `

    <div style="display:flex;gap:8px;flex-wrap:wrap;margin-bottom:4px">

      <span class="dept-tag">${item.dept}</span>

      <span class="${catBadge(item.cat)}">${item.catLabel}</span>

    </div>

    <div class="modal-blood-row">

      <div class="modal-blood-card">

        <div class="big-val" style="color:${rbcColor}">${item.rbcs === "TNS" ? "TNS" : item.rbcs}</div>

        <div class="val-label">${item.rbcs === "TNS" ? "Type & Screen" : "RBC Units"}</div>

      </div>

      ${extraCards}

    </div>

    ${item.conditions ? `<div class="modal-section">

      <div class="modal-section-title">Conditions / Notes</div>

      <p>${item.conditions}</p>

    </div>` : ""}

    ${item.epic.length > 0 ? `<div class="modal-section">

      <div class="modal-section-title">EPIC Procedure Codes</div>

      <ul class="modal-epic-list">

        ${item.epic.map(e => `<li>${e}</li>`).join("")}

      </ul>

    </div>` : ""}

  `;


  document.getElementById("overlay").classList.add("open");

  document.body.style.overflow = "hidden";

}


function closeModal() {

  document.getElementById("overlay").classList.remove("open");

  document.body.style.overflow = "";

}


document.getElementById("modal-close").addEventListener("click", closeModal);

document.getElementById("overlay").addEventListener("click", e => {

  if (e.target === document.getElementById("overlay")) closeModal();

});

document.addEventListener("keydown", e => { if (e.key === "Escape") closeModal(); });

document.getElementById("dept-filters").addEventListener("click", e => {

  const chip = e.target.closest(".chip");

  if (!chip) return;

  document.querySelectorAll("#dept-filters .chip").forEach(c => c.classList.remove("active-dept"));

  chip.classList.add("active-dept");

  activeDept = chip.dataset.dept;

  render();

});


document.getElementById("cat-filters").addEventListener("click", e => {

  const chip = e.target.closest(".chip");

  if (!chip) return;

  document.querySelectorAll("#cat-filters .chip").forEach(c => c.classList.remove("active"));

  chip.classList.add("active");

  activeCat = chip.dataset.cat;

  render();

});
